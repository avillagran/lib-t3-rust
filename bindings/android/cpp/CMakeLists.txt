cmake_minimum_required(VERSION 3.22)
project(t3_jni LANGUAGES CXX)
if(NOT T3_RUST_ARCHIVE OR NOT EXISTS "${T3_RUST_ARCHIVE}")
  message(FATAL_ERROR "Set T3_RUST_ARCHIVE to this ABI's libt3_protocol.a")
endif()
add_library(t3_core STATIC IMPORTED)
set_target_properties(t3_core PROPERTIES IMPORTED_LOCATION "${T3_RUST_ARCHIVE}")
add_library(t3_jni SHARED t3_jni.cpp)
target_include_directories(t3_jni PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../include")
target_compile_features(t3_jni PRIVATE cxx_std_17)
target_compile_options(t3_jni PRIVATE -Wall -Wextra -Werror)
if(ANDROID)
  target_link_libraries(t3_jni PRIVATE t3_core log dl m)
  target_link_options(t3_jni PRIVATE -Wl,-z,max-page-size=16384)
else()
  find_package(JNI REQUIRED)
  target_include_directories(t3_jni PRIVATE ${JNI_INCLUDE_DIRS})
  target_link_libraries(t3_jni PRIVATE t3_core pthread dl m)
endif()
